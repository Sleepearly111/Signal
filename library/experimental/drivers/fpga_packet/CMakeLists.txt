cmake_minimum_required(VERSION 3.20)
project(fpga_packet_protocol LANGUAGES C)

add_library(fpga_packet_protocol STATIC src/fpga_packet_protocol.c)
add_library(SignalFramework::fpga_packet_protocol ALIAS fpga_packet_protocol)
target_include_directories(fpga_packet_protocol PUBLIC include)
target_compile_features(fpga_packet_protocol PUBLIC c_std_11)

if(MSVC)
    target_compile_options(fpga_packet_protocol PRIVATE /W4 /WX)
else()
    target_compile_options(fpga_packet_protocol PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

if(SIGNAL_FRAMEWORK_BUILD_TESTS OR PROJECT_IS_TOP_LEVEL)
    enable_testing()
    add_executable(fpga_packet_protocol_tests tests/test_fpga_packet_protocol.c)
    target_link_libraries(fpga_packet_protocol_tests PRIVATE fpga_packet_protocol)
    target_compile_features(fpga_packet_protocol_tests PRIVATE c_std_11)
    if(MSVC)
        target_compile_options(fpga_packet_protocol_tests PRIVATE /W4 /WX)
    else()
        target_compile_options(fpga_packet_protocol_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
    add_test(NAME DRV-FPGAPKT-001 COMMAND fpga_packet_protocol_tests)
endif()
